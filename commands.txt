Основные команды Git

git init — создаёт новый репозиторий в текущей папке
git clone <url> — копирует удалённый репозиторий на компьютер
git status — показывает состояние файлов (изменённые, новые, в индексе)
git add <файл> — добавляет файл в индекс (область подготовки к коммиту)
git add . — добавляет в индекс все изменения
git commit -m "сообщение" — сохраняет коммит с описанием
git log — показывает историю коммитов
git log --oneline — краткая история коммитов (по одной строке)
git diff — показывает изменения, которые ещё не добавлены в индекс
git branch — показывает список веток
git branch <имя> — создаёт новую ветку
git checkout <ветка> — переключается на другую ветку
git switch <ветка> — современный способ переключения веток
git merge <ветка> — сливает указанную ветку в текущую
git remote add origin <url> — привязывает локальный репозиторий к удалённому
git remote -v — показывает привязанные удалённые репозитории
git push -u origin main — отправляет коммиты на удалённый репозиторий
git pull — забирает и применяет изменения с удалённого репозитория
git fetch — загружает изменения с сервера без слияния
git config --global user.name "Имя" — задаёт имя автора коммитов
git config --global user.email "почта" — задаёт почту автора коммитов
